# Сервисный аккаунт для register_private_dns.py.
# Команды выполняет администратор каталога, не сам новый аккаунт.
# Подставьте свои значения. Идентификатор зоны смотрите так: yc dns zone list

FOLDER_ID=<каталог_с_серверами_BareMetal>
DNS_ZONE_ID=<идентификатор_зоны_Cloud_DNS>
KEY_FILE=$HOME/.config/yc/baremetal-dns-key.json

yc iam service-account create \
  --name baremetal-dns-sync \
  --folder-id "$FOLDER_ID" \
  --description "Синхронизация приватных адресов BareMetal в Cloud DNS"

SA_ID=$(yc iam service-account get baremetal-dns-sync --folder-id "$FOLDER_ID" --jq .id)

# Чтение серверов и назначенных им адресов.
# baremetal.auditor описан как доступ к метаданным и для этого скрипта недостаточен.
# baremetal.viewer не позволяет включать, останавливать и переустанавливать серверы.
yc resource-manager folder add-access-binding "$FOLDER_ID" \
  --role baremetal.viewer \
  --service-account-id "$SA_ID"

# Создание и замена A-записей только в этой зоне.
# Более узкой роли на изменение записей нет: dns.viewer зону только читает.
# Роль назначена зоне, а не каталогу: dns.editor на каталоге ещё и создаёт и удаляет зоны.
yc dns zone add-access-binding "$DNS_ZONE_ID" \
  --role dns.editor \
  --service-account-id "$SA_ID"

mkdir -p -- "$(dirname "$KEY_FILE")"
yc iam key create \
  --service-account-id "$SA_ID" \
  --output "$KEY_FILE"
chmod 600 "$KEY_FILE"

# Проверка без изменения зоны. --zone-id не требует права читать остальные зоны каталога.
python3 register_private_dns.py \
  --sa-key-file "$KEY_FILE" \
  --folder-id "$FOLDER_ID" \
  --zone-id "$DNS_ZONE_ID" \
  --prefix one \
  --prefix two \
  --dry-run
